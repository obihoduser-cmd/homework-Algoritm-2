#include <iostream>
#include <vector>
#include <chrono>
using namespace std;
using namespace chrono;

template <typename T>
class Cache {
private:
    vector<pair<int, T>> data;

public:
    // Добавление или изменение значения
    void put(int key, T value) {
        for (auto& item : data) {
            if (item.first == key) {
                item.second = value;
                return;
            }
        }

        data.push_back({key, value});
    }

    // Получение значения
    bool get(int key, T& result) const {
        for (const auto& item : data) {
            if (item.first == key) {
                result = item.second;
                return true;
            }
        }

        return false;
    }

    // Проверка наличия ключа
    bool contains(int key) const {
        for (const auto& item : data) {
            if (item.first == key) {
                return true;
            }
        }

        return false;
    }

    // Очистка кэша
    void clear() {
        data.clear();
    }

    // Вывод всех элементов
    void print() const {
        for (const auto& item : data) {
            cout << "Ключ: " << item.first
                 << ", значение: " << item.second << endl;
        }
    }
};

// Измерение времени выполнения
template <typename Func>
long long measureTime(Func func) {
    auto start = high_resolution_clock::now();

    func();

    auto end = high_resolution_clock::now();

    return duration_cast<microseconds>(end - start).count();
}

int main() {
    setlocale(LC_ALL, "ru");

    Cache<int> cache;

    // Добавление элементов
    cache.put(1, 10);
    cache.put(2, 20);
    cache.put(3, 30);

    cout << "Содержимое кэша:" << endl;
    cache.print();

    // Получение значения
    int result;

    if (cache.get(2, result)) {
        cout << "\nЗначение по ключу 2: "
             << result << endl;
    } else {
        cout << "Ключ не найден" << endl;
    }

    // Проверка наличия ключа
    cout << "\nЕсть ключ 3? "
         << (cache.contains(3) ? "Да" : "Нет") << endl;

    // Измерение времени выполнения операций
    const int N = 5000;

    long long putTime = measureTime([&]() {
        for (int i = 0; i < N; i++) {
            cache.put(i + 100, i);
        }
    });

    long long getTime = measureTime([&]() {
        for (int i = 0; i < N; i++) {
            cache.get(i + 100, result);
        }
    });

    long long containsTime = measureTime([&]() {
        for (int i = 0; i < N; i++) {
            cache.contains(i + 100);
        }
    });

    long long clearTime = measureTime([&]() {
        cache.clear();
    });

    cout << "\nВремя выполнения операций (" << N
         << " вызовов):" << endl;

    cout << "put: " << putTime << " мкс" << endl;
    cout << "get: " << getTime << " мкс" << endl;
    cout << "contains: " << containsTime << " мкс" << endl;
    cout << "clear: " << clearTime << " мкс" << endl;

    return 0;
}

**Контрольные вопросы:**
1. Что такое О-нотация (Big O) и зачем она нужна?
О-нотация — это способ описания того, как растёт время выполнения или объём памяти алгоритма при увеличении размера входных данных.
2. Какая сложность у доступа к элементу массива по индексу? Почему?
\(O(1)\) — константная сложность, потому что элемент можно получить напрямую по индексу, независимо от размера массива.
3. Какая сложность у бинарного поиска? Почему?
\(O(\log n)\), потому что на каждом шаге область поиска делится пополам. Массив должен быть отсортирован.
4. Что означает сложность \(O(n \log n)\)? Приведите пример алгоритма.
Это означает, что время выполнения растёт примерно пропорционально \(n \log n\). Пример — сортировка слиянием.
5. Что такое амортизированный анализ? Приведите пример.
Амортизированный анализ оценивает среднюю стоимость операции на длинной последовательности операций, учитывая и редкие дорогие операции. Например, добавление элементов в динамический массив обычно имеет амортизированную сложность \(O(1)\).
6. Какая сложность у двух вложенных циклов по n итераций каждый?
\(O(n^2)\), потому что для каждой из \(n\) итераций внешнего цикла внутренний выполняется \(n\) раз: всего \(n \cdot n = n^2\) итераций.
7. Какая сложность у рекурсивного вычисления чисел Фибоначчи?
(O(2^n)) — экспоненциальная сложность для обычного рекурсивного алгоритма без запоминания результатов. Он многократно вычисляет одни и те же значения.
8. Что такое шаблоны функций? Как их объявить и использовать?
Шаблоны функций позволяют создавать функции, работающие с разными типами данных.
template <typename T>
T sum(T a, T b) {
    return a + b;
}

int x = sum(2, 3);
double y = sum(2.5, 3.5);
9. Что такое шаблоны классов? Как их объявить и использовать?
Шаблоны классов позволяют создавать классы, работающие с разными типами данных.
template <typename T>
class Box {
public:
    T value;
};

Box<int> a;
Box<double> b;
10. В чём преимущество шаблонов перед перегрузкой функций?
Шаблоны позволяют написать один универсальный код для разных типов данных вместо создания отдельной функции для каждого типа.
11. Как указать, что шаблонный параметр — тип?
С помощью ключевых слов typename или class.
template <typename T>
12. Можно ли создать шаблон с несколькими параметрами типов?
Да. Например:
template <typename T, typename U>
void print(T a, U b) {
    cout << a << " " << b;
}
13. Как провести экспериментальное измерение времени выполнения?
Можно использовать библиотеку <chrono>, зафиксировав время до и после выполнения алгоритма и вычислив разницу.
14. Почему при измерении времени нужно выполнять операцию многократно?
Чтобы уменьшить влияние случайных задержек и получить более надёжную оценку времени выполнения. Результат можно усреднить.
15. Что важнее: сложность алгоритма или константа?
Оба фактора важны. Для больших объёмов данных обычно важнее асимптотическая сложность, а для небольших данных и практически похожих алгоритмов существенную роль могут играть константы и особенности реализации.
